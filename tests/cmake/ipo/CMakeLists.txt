cmake_minimum_required(VERSION 3.28)
project(ipo_consumer LANGUAGES CXX)
enable_testing()

set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_DEBUG OFF)
set(CMAKE_CONFIGURATION_TYPES Debug Release RelWithDebInfo MinSizeRel DebugOpt)
set(UNI20_ENABLE_LTO ON)
set(expected_optimized OFF)
set(expected_inherited OFF)
if(ipo_case STREQUAL "enabled" OR ipo_case STREQUAL "inherited")
  # Independently establish whether this toolchain can build the IPO fixture.
  include(CheckIPOSupported)
  check_ipo_supported(RESULT fixture_ipo_supported LANGUAGES CXX)
endif()
if(ipo_case STREQUAL "enabled")
  set(expected_optimized ${fixture_ipo_supported})
elseif(ipo_case STREQUAL "inherited")
  set(UNI20_ENABLE_LTO OFF)
  set(expected_inherited ${fixture_ipo_supported})
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ${expected_inherited})
elseif(ipo_case STREQUAL "disabled")
  set(UNI20_ENABLE_LTO OFF)
elseif(ipo_case STREQUAL "debug")
  set(CMAKE_BUILD_TYPE Debug)
  set(CMAKE_CONFIGURATION_TYPES Debug)
elseif(NOT ipo_case STREQUAL "unsupported")
  message(FATAL_ERROR "Unknown IPO fixture case: ${ipo_case}")
endif()

# Fail if skipped paths even load the probe. The unsupported case supplies a
# deterministic failure independent of the compiler installed on the machine.
if(NOT ipo_case STREQUAL "enabled")
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/probe")
endif()

function(assert_ipo target suffix expected)
  get_target_property(actual ${target} INTERPROCEDURAL_OPTIMIZATION${suffix})
  if((actual AND NOT expected) OR (expected AND NOT actual))
    message(FATAL_ERROR "${target} IPO${suffix}: expected ${expected}, got ${actual}")
  endif()
endfunction()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/value.cpp" "int value() { return 42; }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" "int value(); int main() { return value() != 42; }\n")
add_library(earlier_dependency STATIC "${CMAKE_CURRENT_BINARY_DIR}/value.cpp")
add_subdirectory(project)

# Uni20's directory defaults must not change earlier dependency targets or
# leak back into targets created by the enclosing project afterwards.
add_executable(parent_consumer "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
target_link_libraries(parent_consumer PRIVATE earlier_dependency)
foreach(target IN ITEMS earlier_dependency parent_consumer)
  assert_ipo(${target} "" "${expected_inherited}")
  assert_ipo(${target} _RELEASE OFF)
  assert_ipo(${target} _DEBUG OFF)
endforeach()
add_test(NAME parent_ipo_consumer_runs COMMAND parent_consumer)
