include("${UNI20_SOURCE_DIR}/cmake/Uni20IPO.cmake")
add_library(ipo_library STATIC "${PROJECT_BINARY_DIR}/value.cpp")
add_executable(ipo_consumer "${PROJECT_BINARY_DIR}/main.cpp")
target_link_libraries(ipo_consumer PRIVATE ipo_library)
foreach(target IN ITEMS ipo_library ipo_consumer)
  assert_ipo(${target} "" "${expected_inherited}")
  assert_ipo(${target} _DEBUG OFF)
  foreach(config IN LISTS CMAKE_CONFIGURATION_TYPES)
    if(NOT config STREQUAL "Debug")
      string(TOUPPER "${config}" config_upper)
      get_property(multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
      if(multi_config OR config STREQUAL CMAKE_BUILD_TYPE)
        assert_ipo(${target} _${config_upper} "${expected_optimized}")
      endif()
    endif()
  endforeach()
endforeach()
add_test(NAME ipo_consumer_runs COMMAND ipo_consumer)
