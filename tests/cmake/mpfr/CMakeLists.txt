cmake_minimum_required(VERSION 3.28)
project(mpfr_dependency_contract LANGUAGES CXX)
list(APPEND CMAKE_MODULE_PATH "${UNI20_SOURCE_DIR}/cmake")
include(Uni20MPFR)
if(case STREQUAL "missing")
  # Isolate dependency discovery after CMake has found the host toolchain.
  set(CMAKE_FIND_ROOT_PATH "${CMAKE_CURRENT_BINARY_DIR}/empty-root")
  file(MAKE_DIRECTORY "${CMAKE_FIND_ROOT_PATH}")
  set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
  set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
elseif(case STREQUAL "broken")
  set(UNI20_MPFR_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/missing-library.a" CACHE FILEPATH "" FORCE)
endif()
uni20_find_mpfr()
