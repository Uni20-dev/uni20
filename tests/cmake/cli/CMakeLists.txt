cmake_minimum_required(VERSION 3.28)
project(cli_consumer LANGUAGES CXX)

if(cli_case STREQUAL "parent")
  set(UNI20_BUILD_CLI ON CACHE BOOL "" FORCE)
  # Simulate an already-provided package, with no find_package or FetchContent.
  include("${CLI_INCLUDE_FILE}")
  add_library(CLI11::CLI11 INTERFACE IMPORTED)
  set_property(TARGET CLI11::CLI11 PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${CLI_INCLUDE_DIRS}")
else()
  set(UNI20_BUILD_CLI OFF CACHE BOOL "" FORCE)
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_CLI11 TRUE)
set(UNI20_USE_SYSTEM_CLI11 ON CACHE STRING "No fetch or lookup is permitted in this fixture" FORCE)
set(UNI20_ENABLE_MPLAPACK OFF CACHE BOOL "" FORCE)
add_subdirectory("${UNI20_SOURCE_DIR}" uni20)
if(cli_case STREQUAL "disabled")
  if(TARGET uni20_cli OR TARGET CLI11::CLI11)
    message(FATAL_ERROR "Disabled CLI support introduced a CLI target")
  endif()
else()
  if(NOT UNI20_CLI11_SOURCE STREQUAL "parent")
    message(FATAL_ERROR "Parent CLI11 target was not reused")
  endif()
endif()
foreach(target IN ITEMS uni20 uni20_deps uni20_common uni20_core uni20_linalg)
  get_target_property(links ${target} INTERFACE_LINK_LIBRARIES)
  if("${links}" MATCHES "CLI11|uni20_cli")
    message(FATAL_ERROR "CLI dependency leaked into ${target}: ${links}")
  endif()
endforeach()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE uni20_common)
if(cli_case STREQUAL "parent")
  target_link_libraries(consumer PRIVATE uni20_cli)
  target_compile_definitions(consumer PRIVATE TEST_CLI)
endif()
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
