cmake_minimum_required(VERSION 3.28)
project(dependency_ownership LANGUAGES NONE)
include("${UNI20_SOURCE_DIR}/cmake/Uni20Dependencies.cmake")

set(UNI20_FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}/uni20-deps")
set(UNI20_FETCHCONTENT_SOURCE_BASE_DIR "${UNI20_FETCHCONTENT_BASE_DIR}")
set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}/parent-deps")

# Exercise real package resolution, including names that used to be mistaken
# for Uni20-owned directories solely because of their spelling.
foreach(location IN ITEMS staged _deps/package .cmake/third_party/package)
  set(package_dir "${CMAKE_CURRENT_BINARY_DIR}/${location}")
  file(MAKE_DIRECTORY "${package_dir}")
  file(WRITE "${package_dir}/ParentPackageConfig.cmake"
    "set(ParentPackage_FOUND TRUE)\n")
  set(ParentPackage_DIR "${package_dir}" CACHE PATH "Parent-owned package" FORCE)
  set(UNI20_USE_SYSTEM_PARENTPACKAGE ON)
  uni20_add_dependency(NAME ParentPackage TARGET ParentPackage::library)
  if(NOT ParentPackage_DIR STREQUAL package_dir)
    message(FATAL_ERROR "Lost parent-owned package hint: ${package_dir}")
  endif()
endforeach()
_uni20_is_internal_path("${UNI20_FETCHCONTENT_BASE_DIR}/stale/package" owned)
if(NOT owned)
  message(FATAL_ERROR "Uni20-managed stale package paths must still be recognized")
endif()

# FetchContent configures a shared dependency only once. Verify its actual
# targets, not just the parent's unchanged cache text.
set(fixture "${CMAKE_CURRENT_BINARY_DIR}/dependency-source")
file(MAKE_DIRECTORY "${fixture}")
file(WRITE "${fixture}/CMakeLists.txt" [=[
cmake_minimum_required(VERSION 3.28)
project(ParentFeature LANGUAGES NONE)
option(PARENT_FEATURE "Parent requested component" OFF)
option(DEFAULT_FEATURE "Optional component" ON)
option(PARENT_DISABLED "Parent disabled component" ON)
if(PARENT_FEATURE)
  add_library(parent_feature INTERFACE)
endif()
if(DEFAULT_FEATURE)
  message(FATAL_ERROR "Undefined option did not receive Uni20's default")
endif()
if(PARENT_DISABLED)
  message(FATAL_ERROR "Explicit parent OFF was overwritten")
endif()
add_library(parent_dependency INTERFACE)
]=])
set(PARENT_FEATURE ON CACHE BOOL "Parent choice" FORCE)
set(PARENT_DISABLED OFF)
set(FETCHCONTENT_SOURCE_DIR_PARENTFEATURE "${fixture}")
set(UNI20_USE_SYSTEM_PARENTFEATURE OFF)
uni20_add_dependency(NAME ParentFeature TARGET parent_dependency
  DEFAULTS PARENT_FEATURE=OFF PARENT_DISABLED=ON DEFAULT_FEATURE=OFF)
include(FetchContent)
FetchContent_MakeAvailable(ParentFeature)
if(NOT TARGET parent_feature OR NOT PARENT_FEATURE)
  message(FATAL_ERROR "Parent-requested component was suppressed during population")
endif()
