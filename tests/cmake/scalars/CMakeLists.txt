cmake_minimum_required(VERSION 3.28)
project(scalar_configuration_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
list(APPEND CMAKE_MODULE_PATH "${UNI20_SOURCE_DIR}/cmake")
include(Uni20Dependencies)

if(case STREQUAL "missing")
  # Toolchain discovery is finished: isolate only scalar package discovery.
  set(CMAKE_FIND_ROOT_PATH "${CMAKE_CURRENT_BINARY_DIR}/empty-root")
  file(MAKE_DIRECTORY "${CMAKE_FIND_ROOT_PATH}")
  set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
  set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
  set(CMAKE_DISABLE_FIND_PACKAGE_mplapack TRUE)
elseif(case STREQUAL "broken_mpc")
  set(UNI20_MPC_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/missing-mpc.a" CACHE FILEPATH "" FORCE)
elseif(case STREQUAL "no_float80")
  # Compile-only x86 ABI probe; never execute code with the altered long double.
  set(CMAKE_REQUIRED_FLAGS "-mlong-double-64")
endif()

include(Uni20Scalars)
uni20_configure_scalar_providers()
foreach(family IN ITEMS FLOAT80 FLOAT128 MPREAL)
  if(DEFINED expect_${family} AND NOT "${UNI20_HAS_${family}}" STREQUAL "${expect_${family}}")
    message(FATAL_ERROR "${family}: expected ${expect_${family}}, got ${UNI20_HAS_${family}}")
  endif()
  if(UNI20_HAS_MPLAPACK_${family} AND NOT UNI20_BACKEND_MPLAPACK)
    message(FATAL_ERROR "Disabled backend advertised ${family} kernels")
  endif()
endforeach()
if(UNI20_HAS_MPREAL AND NOT TARGET uni20_mpc)
  message(FATAL_ERROR "mpreal must include mpcomplex's MPC dependency")
endif()
